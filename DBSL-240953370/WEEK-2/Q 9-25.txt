8.
select
  s.name as student_name,
  s.dept_name as department_name,
  i.name as advisor_name,
  count(t.course_id) as courses_registered
from student s
left join advisor a
  on s.ID = a.s_ID
left join instructor i
  on a.i_ID = i.ID
left join takes t
  on s.ID = t.ID
group by s.name, s.dept_name, i.name;
